151def sample_cell(bounds: tuple[float, float, float, float], sample_lon: float, sample_lat: float) -> tuple[int, int]: 

152 min_lon, min_lat, max_lon, max_lat = bounds 

153 H, W = GRID_SHAPE 

154 col = int(np.clip((sample_lon - min_lon) / (max_lon - min_lon) * W, 0, W - 1)) 

155 row = int(np.clip((max_lat - sample_lat) / (max_lat - min_lat) * H, 0, H - 1)) 

156 return row, col

158def destination(lat: float, lng: float, distance_km: float, bearing_deg: float) -> tuple[float, float]: 

159 dlat = (distance_km * math.cos(math.radians(bearing_deg))) / (METERS_PER_DEG_LAT / 1000.0) 

160 dlon = (distance_km * math.sin(math.radians(bearing_deg))) / ( 

161 (METERS_PER_DEG_LAT / 1000.0) * math.cos(math.radians(lat)) 

162 ) 

163 return lat + dlat, lng + dlon

165def compute_terrain_bias(lat: float, lng: float) -> list[dict]: 

166 """ 

167 Returns N_BEARINGS {bearing_deg, factor} entries, factor averaging to ~1.0 across 

168 all bearings - relative reshaping multiplier, not absolute shape. 

169 Wind driven ROS formula on client still sets overall size. This just distorts the shape 

170 toward real uphill/fuel-type tendencies 

171 """ 

172 fuel_weights = load_fuel_base_weights() 

173 grids = fetch_terrain_grids(lat, lng) 

174 bounds = grids["bounds"] 

175 

176 raw_factors = [] 

177 for i in range(N_BEARINGS): 

178 bearing = (360.0 / N_BEARINGS) * i 

179 sample_lat, sample_lon = destination(lat, lng, SAMPLE_RADIUS_KM, bearing) 

180 row, col = sample_cell(bounds, sample_lon, sample_lat) 

181 

182 slope_deg = float(grids["slope"][row, col]) 

183 aspect_deg = float(grids["aspect"][row, col]) # convention: direction slope faces (downhill) 

184 landcover_class = int(grids["landcover"][row, col]) 

185 

186 uphill_bearing = (aspect_deg + 180.0) % 360.0 

187 alignment = math.cos(math.radians(bearing - uphill_bearing)) # 1 = uphill, 2 = downhill 

188 slope_factor = 1.0 + 1.2 * math.tan(math.radians(min(slope_deg, 60.0))) * alignment 

189 slope_factor = max(0.3, min(3.0, slope_factor)) 

190 

191 fuel_weight = fuel_weights.get(landcover_class, 0.2) 

192 landcover_factor = 0.4 + 1.2 * fuel_weight 

193 

194 raw_factors.append(slope_factor * landcover_factor) 

195 

196 mean_factors = sum(raw_factors) / len(raw_factors) or 1.0 

197 return [ 

198 {"bearing_deg": (360.0 / N_BEARINGS) * i, "factor": raw_factors[i] / mean_factors} 

199 for i in range(N_BEARINGS) 

200 ]
